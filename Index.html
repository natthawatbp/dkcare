  function renderPaginationControls(startItem, endItem, totalItems, totalPages) {

      info.textContent = '';

    let html = `<li class="page-item ${currentPage === 1 ? 'disabled' : ''}"><button class="page-link" onclick="changePage(${currentPage - 1})">ก่อนหน้า</button></li>`;

    

    let startPage = Math.max(1, currentPage - 2);

    let endPage = Math.min(totalPages, startPage + 4);

    if (endPage - startPage < 4) startPage = Math.max(1, endPage - 4);


    for (let i = startPage; i <= endPage; i++) {

      html += `<li class="page-item ${currentPage === i ? 'active' : ''}"><button class="page-link" onclick="changePage(${i})">${i}</button></li>`;

    }


    html += `<li class="page-item ${currentPage === totalPages ? 'disabled' : ''}"><button class="page-link" onclick="changePage(${currentPage + 1})">ถัดไป</button></li>`;

    ul.innerHTML = html;

  }

  function evaluateTemperature(v) {

    if (v > 40.5) return { score: 1, label: '🔴 ไข้สูงมาก', color: '#dc2626' };

    if (v >= 39.5) return { score: 1, label: '🔴 ไข้สูง', color: '#ef4444' };

    if (v >= 38.4) return { score: 1, label: '🟠 ไข้ปานกลาง', color: '#f97316' };

    if (v >= 37.6) return { score: 1, label: '🟡 ไข้ต่ำ', color: '#eab308' };

    if (v < 35) return { score: 1, label: '🔴 ตัวเย็นผิดปกติ', color: '#dc2626' };

    return { score: 0, label: '🟢 ปกติ', color: '#16a34a' };

  }


  function evaluatePulse(age, v) {

    let min = 60, max = 100;

    if (age <= 0.083) { min = 120; max = 160; } else if (age < 1) { min = 80; max = 140; } else if (age <= 5) { min = 65; max = 120; }

    return rangeResult(v, min, max);

  }


  function evaluateRespiration(age, v) {

    let min = 12, max = 20;

    if (age <= 0.1667) { min = 30; max = 60; } else if (age < 1) { min = 25; max = 50; } else if (age <= 3) { min = 20; max = 40; } else if (age <= 5) { min = 22; max = 34; } else if (age <= 12) { min = 18; max = 30; }

    return rangeResult(v, min, max);

  }


  function rangeResult(value, min, max) {

    if (!isFinite(value)) return null;

    if (value >= min && value <= max) return { score: 0, label: `🟢 ปกติ (${min}-${max})`, color: '#16a34a' };

    return { score: 1, label: `🔴 ผิดปกติ (${min}-${max})`, color: '#dc2626' };

  }


  function evaluateBP(age, sys, dia) {

    if (!isFinite(sys) || !isFinite(dia)) return null;

    let maxSys = 140, maxDia = 90;

    if (age < 3) { maxSys = 90; maxDia = 60; } else if (age <= 6) { maxSys = 110; maxDia = 70; } else if (age <= 17) { maxSys = 120; maxDia = 80; } else if (age >= 65) { maxSys = 160; maxDia = 95; }

    

    if (sys <= maxSys && dia <= maxDia) return { score: 0, label: '🟢 ปกติ', color: '#16a34a' };

    return { score: 1, label: '🔴 สูงกว่าค่าเกณฑ์', color: '#dc2626' };

  }


  function evaluateSugar(value, condition) {

    if (!isFinite(value)) return null;

    if (condition === 'FASTING') {

      if (value < 100) return { score: 0, label: '🟢 ปกติ', color: '#16a34a' };

      if (value <= 125) return { score: 1, label: '🟡 ภาวะเสี่ยงเป็นเบาหวาน', color: '#eab308' };

      return { score: 1, label: '🔴 สูง ควรประเมินเพิ่มเติม', color: '#dc2626' };

    }

    if (value <= 140) return { score: 0, label: '🟢 ปกติ', color: '#16a34a' };

    if (value < 180) return { score: 1, label: '🟡 ควรเฝ้าระวัง', color: '#eab308' };

    return { score: 1, label: '🔴 สูง ควรประเมินเพิ่มเติม', color: '#dc2626' };

  }


  function evaluateSpO2(v) {

    if (!isFinite(v)) return null;

  function renderMap(data) {

    setTimeout(function() {

      if (!map) {

        map = L.map('map').setView([18.7883, 98.9853], 10);

        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', { attribution: '&copy; OpenStreetMap contributors' }).addTo(map);

      }

      map.eachLayer(function(layer) { if (layer instanceof L.Marker) map.removeLayer(layer); });


      const valid = data.filter(p => isFinite(p.lat) && isFinite(p.lng) && p.lat !== 0 && p.lng !== 0);


      valid.forEach(function(p) {

        const color = getStatusColor(p.status);

        const icon = L.divIcon({

          className: 'dk-map-marker',

          html: `<div style="width:26px; height:26px; border-radius:50%; background:${color}; border:3px solid white; box-shadow:0 2px 8px rgba(0,0,0,.3);"></div>`,

          iconSize: [26,26], iconAnchor: [13,13]

        });

        L.marker([p.lat, p.lng], {icon: icon})

          .addTo(map)

          .bindPopup(`<strong>${escapeHtml(p.name)}</strong><br>สถานะ: ${escapeHtml(p.status)}`);

      });


      if (valid.length) {

        const bounds = L.latLngBounds(valid.map(p => [p.lat, p.lng]));

        map.fitBounds(bounds, {padding: [30,30]});

      }

      map.invalidateSize();

    }, 200);

  }

      }


      return `

        <tr>

          <td><strong>${escapeHtml(u.FullName)}</strong><br><small class="text-muted">${escapeHtml(u.Phone || '')}</small></td>

          <td>${escapeHtml(u.Position)}</td>

          <td>${escapeHtml(u.CreatedAt)}</td>

      `;

    }).join('');

  }

  function showAlert(title, message, type, callback) {

    document.getElementById('alertTitle').textContent = title;

    document.getElementById('alertMessage').innerHTML = message;

    

    let icon = '<i class="bi bi-info-circle text-primary"></i>';

    if (type === 'success') icon = '<i class="bi bi-check-circle-fill text-success"></i>';

    else if (type === 'danger') icon = '<i class="bi bi-x-circle-fill text-danger"></i>';

    else if (type === 'warning') icon = '<i class="bi bi-exclamation-triangle-fill text-warning"></i>';

    

    document.getElementById('alertIcon').innerHTML = icon;

    const modalElement = document.getElementById('alertModal');


    if (callback) {

      modalElement.addEventListener('hidden.bs.modal', function handler() {

        modalElement.removeEventListener('hidden.bs.modal', handler);

        callback();

      });

    }

    alertModal.show();

  }


  function showConfirm(message, callback) {

    document.getElementById('confirmMessage').innerHTML = message;

    const button = document.getElementById('confirmButton');

    const newButton = button.cloneNode(true);

    button.parentNode.replaceChild(newButton, button);

    

    newButton.addEventListener('click', function() {

      confirmModal.hide();

      callback();

    });

    confirmModal.show();

  }


  function statusBadge(status) {

    let cls = 'status-normal';

    if (status === 'เฝ้าระวัง') cls = 'status-watch';

  }

  function escapeHtml(value) {

    if (value === null || value === undefined) return '';

    return String(value).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;').replace(/'/g, '&#039;');

  }


  function escapeJs(value) {

    return String(value || '').replace(/\\/g, '\\\\').replace(/'/g, "\\'");

  }
